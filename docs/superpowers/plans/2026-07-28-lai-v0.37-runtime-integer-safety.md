# LAI v0.37 Runtime Integer Safety Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every existing LAI i32 operation deterministic compile-time or runtime safety, while keeping one language mode and preserving the experimental LLVM capability boundary.

**Architecture:** `lai_int.py` becomes the shared source of truth for pure-static i32 evaluation and precise failures. The checker converts those failures to source diagnostics; the C backend uses collision-free generated runtime helpers for dynamic values and a range-aware loop advance helper; LLVM continues accepting only its existing pure-static integer subset after the shared checker proves it safe.

**Tech Stack:** Python 3, `unittest`, generated C17-compatible source, `clang`, textual LLVM IR, PowerShell, Git.

**Status:** Planned.

## Global Constraints

- Current base commit before plan creation: `98f0eb9e0624b6aa549ca8f90e849f34a86af47b`.
- There is exactly one integer mode: checked i32 safety. Do not add a performance, unchecked, wrapping, debug, or release switch.
- LAI `int` remains `-2147483648..2147483647`; do not add syntax or widen the type.
- Pure-static failures are compile errors in `line N: <reason>` form.
- Dynamic failures write exactly `LAI runtime error: line N: <reason>` to `stderr`, then call `exit(EXIT_FAILURE)`.
- Error reasons are exactly: `division by zero`, `modulo by zero`, `integer addition overflow`, `integer subtraction overflow`, `integer multiplication overflow`, `integer unary negation overflow`, `integer division overflow`, `integer modulo overflow`, and `for step must be greater than 0`.
- `INT_MIN / -1` and `INT_MIN % -1` are overflow errors; neither backend may wrap or invoke C/LLVM undefined behavior.
- Pure-static safe arithmetic may remain readable raw C. Arithmetic containing a dynamic value uses checked helpers; v0.37 does not add range analysis.
- `for` start, end, and step evaluate once before the loop. A dynamic non-positive step fails even for an empty range.
- Positive range iteration ends normally when the mathematical next item crosses the fixed end; `through INT_MAX` must not produce a fake post-range overflow.
- Runtime helper and temporary names must not collide with any user function, parameter, local, or loop variable; do not reserve a new LAI source prefix.
- Preserve `and` / `or` runtime short-circuiting: a checked arithmetic helper in an unexecuted operand must not run.
- Do not add Python dependencies, a new AST node, a new source token, or a general IR layer.
- LLVM remains limited to empty `main` and top-level pure integer `print`; do not add variables, assignments, functions, strings, booleans, comparisons, or control flow.
- Stage only intended source, tests, examples, and docs. Never stage `build/`, `examples/build/`, executables, credentials, or unrelated user changes.
- After every task: run its focused tests and the full suite, review the exact diff, commit one logical change, and push `main`.

- [x] **Step 4: Implement the i32-aware evaluator**

In `lai_int.py`, retain the existing constants and compatibility evaluator, then add:

```python
class StaticIntError(ValueError):
    pass


def _require_i32(value: int, operation: str) -> int:
    if not I32_MIN <= value <= I32_MAX:
        raise StaticIntError(f"integer {operation} overflow")
    return value


def evaluate_static_i32(expr) -> int | None:
    if isinstance(expr, IntExpr):
        if type(expr.value) is not int:
            return None
        if not 0 <= expr.value <= I32_MAX:
            raise StaticIntError(f"integer literal out of i32 range: {expr.value}")
        return expr.value
    if isinstance(expr, GroupExpr):
        return evaluate_static_i32(expr.value)
    if isinstance(expr, UnaryExpr):
        if expr.operator == "-" and is_i32_min_magnitude_expr(expr.operand):
            return I32_MIN
        if expr.operator not in {"+", "-"}:
            return None
        value = evaluate_static_i32(expr.operand)
        if value is None:
            return None
        if expr.operator == "+":
            return value
        return _require_i32(-value, "unary negation")
    if isinstance(expr, AddExpr):
        if not expr.terms:
            return None
        result = evaluate_static_i32(expr.terms[0])
        is_dynamic = result is None
        for term in expr.terms[1:]:
            value = evaluate_static_i32(term)
            if value is None:
                is_dynamic = True
            elif not is_dynamic:
                result = _require_i32(result + value, "addition")
        return None if is_dynamic else result
    if isinstance(expr, SubtractExpr):
        left = evaluate_static_i32(expr.left)
        right = evaluate_static_i32(expr.right)
        if left is None or right is None:
            return None
        return _require_i32(left - right, "subtraction")
    if isinstance(expr, MultiplyExpr):
        if not expr.factors:
            return None
        result = evaluate_static_i32(expr.factors[0])
        is_dynamic = result is None
        for factor in expr.factors[1:]:
            value = evaluate_static_i32(factor)
            if value is None:
                is_dynamic = True
            elif not is_dynamic:
                result = _require_i32(result * value, "multiplication")
        return None if is_dynamic else result
    if isinstance(expr, DivideExpr):
        left = evaluate_static_i32(expr.left)
        right = evaluate_static_i32(expr.right)
        if right == 0:
            raise StaticIntError("division by zero")
        if left is None or right is None:
            return None
        if left == I32_MIN and right == -1:
            raise StaticIntError("integer division overflow")
        return _truncate_toward_zero(left, right)
    if isinstance(expr, ModuloExpr):
        left = evaluate_static_i32(expr.left)
        right = evaluate_static_i32(expr.right)
        if right == 0:
            raise StaticIntError("modulo by zero")
        if left is None or right is None:
            return None
        if left == I32_MIN and right == -1:
            raise StaticIntError("integer modulo overflow")
        quotient = _truncate_toward_zero(left, right)
        return left - quotient * right
    return None
```

The explicit `is_dynamic` state ensures a dynamic accumulated prefix cannot become
static again when a later term is a literal. Child errors still propagate, and every
fully known left prefix is checked before accumulation continues.

```c
_Static_assert(INT_MIN == (-2147483647 - 1) && INT_MAX == 2147483647,
               "LAI requires 32-bit int");

static void PREFIX_runtime_error(int line, const char* message) {
    fprintf(stderr, "LAI runtime error: line %d: %s\n", line, message);
    exit(EXIT_FAILURE);
}

static int PREFIX_i32_add(int left, int right, int line) {
    int64_t result = (int64_t)left + (int64_t)right;
    if (result < INT_MIN || result > INT_MAX) {
        PREFIX_runtime_error(line, "integer addition overflow");
    }
    return (int)result;
}

static int PREFIX_i32_subtract(int left, int right, int line) {
    int64_t result = (int64_t)left - (int64_t)right;
    if (result < INT_MIN || result > INT_MAX) {
        PREFIX_runtime_error(line, "integer subtraction overflow");
    }
    return (int)result;
}

static int PREFIX_i32_multiply(int left, int right, int line) {
    int64_t result = (int64_t)left * (int64_t)right;
    if (result < INT_MIN || result > INT_MAX) {
        PREFIX_runtime_error(line, "integer multiplication overflow");
    }
    return (int)result;
}

static int PREFIX_i32_negate(int value, int line) {
    if (value == INT_MIN) {
        PREFIX_runtime_error(line, "integer unary negation overflow");
    }
    return -value;
}

static int PREFIX_i32_divide(int left, int right, int line) {
    if (right == 0) {
        PREFIX_runtime_error(line, "division by zero");
    }
    if (left == INT_MIN && right == -1) {
        PREFIX_runtime_error(line, "integer division overflow");
    }
    return left / right;
}

static int PREFIX_i32_modulo(int left, int right, int line) {
    if (right == 0) {
        PREFIX_runtime_error(line, "modulo by zero");
    }
    if (left == INT_MIN && right == -1) {
        PREFIX_runtime_error(line, "integer modulo overflow");
    }
    return left % right;
}

static int PREFIX_require_positive_step(int value, int line) {
    if (value <= 0) {
        PREFIX_runtime_error(line, "for step must be greater than 0");
    }
    return value;
}

static int PREFIX_for_advance(
    int* current, int step, int end, int inclusive
) {
    int64_t next = (int64_t)(*current) + (int64_t)step;
    if (inclusive ? next > end : next >= end) {
        return 0;
    }
    *current = (int)next;
    return 1;
}
```
