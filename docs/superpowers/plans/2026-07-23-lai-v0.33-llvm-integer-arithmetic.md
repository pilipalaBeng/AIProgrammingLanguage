# LAI v0.33 LLVM Integer Arithmetic Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend the experimental textual LLVM backend so existing integer
arithmetic and parenthesized expressions can be printed, compiled, and run.

**Architecture:** Keep the shared lexer, parser, AST, checker, backend descriptor,
and clang runner unchanged. Replace the literal-only LLVM body generation with a
small stateful `_LlvmMainEmitter` that recursively lowers expressions to unique
SSA-style temporaries while carrying a synchronized signed `i32` value used only
to reject LLVM undefined division and remainder cases.

**Tech Stack:** Python 3.12+, standard-library `unittest`, textual LLVM IR,
local LLVM/Clang 22.1.8 targeting Windows.

## Global Constraints

- Do not add or change LAI source syntax in v0.33.
- Keep the complete C backend as the default and preserve its generated output.
- Keep `generate_llvm(program)` defensive and keep `compile_source(..., LLVM_BACKEND)` at exactly one semantic-checker invocation.
- Support LLVM lowering only for `IntExpr`, `AddExpr`, `SubtractExpr`, `MultiplyExpr`, `DivideExpr`, `ModuloExpr`, and `GroupExpr` inside top-level `PrintStmt.value`.
- Do not add variables, assignments, comparisons, booleans, strings, control flow, user functions, `llvmlite`, or a LAI-specific lower-level IR.
- Validate every `IntExpr` payload with `type(value) is int` and range `0..2147483647` before placing it in IR.
- Emit real `add`, `sub`, `mul`, `sdiv`, and `srem` instructions; synchronized values must not replace expressions with folded constants.
- Emit `add`, `sub`, and `mul` without `nsw`, `nuw`, or `exact`; synchronized values use signed 32-bit two's-complement wrapping.
- Reject computed zero divisors with `line N: division by zero` or `line N: modulo by zero`.
- Reject `INT_MIN / -1` and `INT_MIN % -1` with explicit LLVM backend overflow errors.
- Preserve v0.32 empty-main, integer-literal print, capability-error, clang-build, and CLI behavior.
- Use TDD before every production-code change.
- Commit each completed task; push after v0.33 has passed task reviews, final review, and full verification.

Define signed `i32` wrapping and the emitter before `generate_llvm`:

```python
_I32_MIN = -2147483648
_I32_MAX = 2147483647
_I32_MODULUS = 4294967296


def _wrap_i32(value: int) -> int:
    return ((value - _I32_MIN) % _I32_MODULUS) + _I32_MIN


class _LlvmMainEmitter:
    def __init__(self) -> None:
        self.body_lines: list[str] = []
        self.next_value_index = 0

    def emit_print(self, statement: PrintStmt, print_index: int) -> None:
        operand, _ = self.lower_int_expr(statement.value, statement.line)
        self.body_lines.append(
            f"  %print{print_index} = call i32 (ptr, ...) "
            f"@printf(ptr @.fmt.int, i32 {operand})"
        )

    def lower_int_expr(self, expr, line: int) -> tuple[str, int]:
        if isinstance(expr, IntExpr):
            return self._lower_int_literal(expr, line)
        if isinstance(expr, GroupExpr):
            return self.lower_int_expr(expr.value, line)
        if isinstance(expr, AddExpr):
            return self._lower_chain(expr.terms, "add", line)
        if isinstance(expr, SubtractExpr):
            left_operand, left_value = self.lower_int_expr(expr.left, line)
            right_operand, right_value = self.lower_int_expr(expr.right, line)
            result = self._emit_binary("sub", left_operand, right_operand)
            return result, _wrap_i32(left_value - right_value)
        if isinstance(expr, MultiplyExpr):
            return self._lower_chain(expr.factors, "mul", line)
        raise LaiCompileError(
            f"line {line}: LLVM backend does not support "
            f"{type(expr).__name__} yet"
        )

    def _lower_int_literal(self, expr: IntExpr, line: int) -> tuple[str, int]:
        if type(expr.value) is not int:
            raise LaiCompileError(
                f"line {line}: LLVM backend int literal must be int, got "
                f"{type(expr.value).__name__}"
            )
        if not 0 <= expr.value <= _I32_MAX:
            raise LaiCompileError(
                f"line {line}: LLVM backend int literal out of i32 range: "
                f"{expr.value}"
            )
        return str(expr.value), expr.value

    def _lower_chain(
        self, expressions: list, opcode: str, line: int
    ) -> tuple[str, int]:
        operand, value = self.lower_int_expr(expressions[0], line)
        for expression in expressions[1:]:
            right_operand, right_value = self.lower_int_expr(expression, line)
            operand = self._emit_binary(opcode, operand, right_operand)
            if opcode == "add":
                value = _wrap_i32(value + right_value)
            else:
                value = _wrap_i32(value * right_value)
        return operand, value

    def _emit_binary(
        self, opcode: str, left_operand: str, right_operand: str
    ) -> str:
        result = f"%value{self.next_value_index}"
        self.next_value_index += 1
        self.body_lines.append(
            f"  {result} = {opcode} i32 {left_operand}, {right_operand}"
        )
        return result
```

```python
def _lower_division_like(
    self,
    left_expr,
    right_expr,
    opcode: str,
    operation_name: str,
    line: int,
) -> tuple[str, int]:
    left_operand, left_value = self.lower_int_expr(left_expr, line)
    right_operand, right_value = self.lower_int_expr(right_expr, line)
    if right_value == 0:
        error_name = "division" if opcode == "sdiv" else "modulo"
        raise LaiCompileError(f"line {line}: {error_name} by zero")
    if left_value == _I32_MIN and right_value == -1:
        raise LaiCompileError(
            f"line {line}: LLVM backend signed {operation_name} overflow"
        )

    result = self._emit_binary(opcode, left_operand, right_operand)
    quotient = _truncate_toward_zero(left_value, right_value)
    if opcode == "sdiv":
        return result, quotient
    return result, left_value - quotient * right_value
```

Add the module helper:

```python
def _truncate_toward_zero(left: int, right: int) -> int:
    quotient = abs(left) // abs(right)
    if (left < 0) != (right < 0):
        return -quotient
    return quotient
```
